# vm1/docker-compose.yml
# ==============================================================================
# docker-compose.yml — ВМ №1: Периметр, шлюзы, observability, зеркала
# ==============================================================================
#
# Состав:
#   - nginx (ingress / SSE-proxy)
#   - oauth2-proxy + Keycloak (OIDC / LDAP-федерация)
#   - PostgreSQL — Auth & Observability (только базы Keycloak и Langfuse)
#   - Redis — Langfuse (noeviction, ACL langfuse_svc)
#   - Langfuse web + worker (v4) + ClickHouse (26.8 LTS)
#   - PyPI mirror (доступ только с ВМ №2)
#   - AG-UI Gateway (FastAPI)
#
# ВАЖНО перед первым стартом:
#   1. Скопировать .env.example в .env и заполнить значения.
#   2. Запустить ./render-configs.sh → создаст rendered/realm-export.json.
#   3. Поместить корпоративный TLS-сертификат в nginx/certs/
#      (SAN должен покрывать: factory.corp.example.com, langfuse.corp.example.com).
#   4. Поместить статическую сборку SPA в spa/dist/.
#   5. Задать VM1_HOST_IP и VM2_HOST_IP в .env.
#
# ==============================================================================
name: factory-vm1
services:
  # ============================================================================
  # INGRESS / SSE-PROXY
  # ============================================================================
  nginx:
    image: nginx:1.31.9-alpine  # pin digest from mirror
    container_name: vm1-nginx
    restart: unless-stopped
    ports:
      - "443:443"
      - "80:80"
    command: ["nginx", "-c", "/etc/nginx-runtime/nginx.conf", "-g", "daemon off;"]
    volumes:
      - nginx-runtime-config:/etc/nginx-runtime:ro
      - nginx-tls:/etc/nginx/certs:ro
      - spa-dist:/usr/share/nginx/html:ro
      - nginx-cache:/var/cache/nginx
      - nginx-logs:/var/log/nginx
    environment:
      TZ: ${TZ:-Asia/Yekaterinburg}
    depends_on:
      oauth2-proxy:
        condition: service_started
      gateway:
        condition: service_started
    networks:
      frontend:
        aliases:
          - langfuse.corp.example.com
    healthcheck:
      test: ["CMD", "wget", "--spider", "-q", "http://127.0.0.1:80/health"]
      interval: 30s
      timeout: 5s
      retries: 3
    logging:
      driver: json-file
      options:
        max-size: "50m"
        max-file: "4"

  # ============================================================================
  # AUTH-PROXY (oauth2-proxy + Keycloak OIDC)
  # Секреты передаются через environment (OAUTH2_PROXY_* — нативная поддержка),
  # а НЕ через ${...} в cfg — oauth2-proxy не раскрывает переменные в конфиге.
  # ============================================================================
  oauth2-proxy:
    image: quay.io/oauth2-proxy/oauth2-proxy:v7.6.0  # pin digest from mirror
    container_name: vm1-oauth2-proxy
    restart: unless-stopped
    command:
      - --config=/etc/oauth2-proxy-config/oauth2-proxy.cfg
      - --cookie-secret=${OAUTH2_PROXY_COOKIE_SECRET}
      - --provider-ca-file=/etc/oauth2-proxy-certs/ca.crt
      - --insecure-oidc-skip-issuer-verification
    environment:
      # Нативные переменные окружения (имеют приоритет над cfg)
      OAUTH2_PROXY_CLIENT_ID: ${OAUTH2_PROXY_CLIENT_ID}
      OAUTH2_PROXY_CLIENT_SECRET: ${OAUTH2_PROXY_CLIENT_SECRET}
      OAUTH2_PROXY_COOKIE_SECRET: ${OAUTH2_PROXY_COOKIE_SECRET}
      OAUTH2_PROXY_OIDC_ISSUER_URL: http://keycloak:8080/auth/realms/factory
      TZ: ${TZ:-Asia/Yekaterinburg}
    volumes:
      - oauth2-proxy-config:/etc/oauth2-proxy-config:ro
      - nginx-tls:/etc/oauth2-proxy-certs:ro
    depends_on:
      keycloak:
        condition: service_healthy
    networks:
      - frontend
    # OAuth2-proxy image is distroless; probe it from host/network diagnostics.
    logging:
      driver: json-file
      options:
        max-size: "50m"
        max-file: "4"

  # ============================================================================
  # IdP / LDAP-FEDERATION (Keycloak)
  # Path-based: KC_HTTP_RELATIVE_PATH=/auth, публично через nginx /auth/
  # Realm импортируется из rendered/ при первом запуске (--import-realm).
  # ============================================================================
  keycloak:
    image: quay.io/keycloak/keycloak:26.0  # pin digest from mirror
    container_name: vm1-keycloak
    restart: unless-stopped
    command:
      - start
      - --proxy-headers=xforwarded
      - --hostname-strict=false
      - --import-realm
    environment:
      KC_DB: postgres
      KC_DB_URL: jdbc:postgresql://postgres-auth:5432/keycloak
      KC_DB_USERNAME: ${KEYCLOAK_DB_USER}
      KC_DB_PASSWORD: ${KEYCLOAK_DB_PASSWORD}
      KC_HTTP_ENABLED: "true"
      KC_HEALTH_ENABLED: "true"
      KC_HTTP_RELATIVE_PATH: /auth

#!/usr/bin/env bash
# ==============================================================================
# render-configs.sh — Рендеринг конфигурационных файлов ВМ №1
# Production Blueprint v5.2
# ==============================================================================
# Назначение:
#   Рендеринг шаблонов конфигов, которые не поддерживают переменные окружения
#   нативно. На ВМ №1 это файл репрезентации Keycloak (realm-export.json).
#
# Что делает:
#   1. Подставляет переменные окружения из .env в шаблон через envsubst.
#   2. Вычищает служебные ключи с префиксом "_" из репрезентации
#      (_ldap_template, _comment_* и т.д.), которые находятся вне схемы
#      и могут вызвать отказ импорта репрезентации.
#   3. Сохраняет результат в ./rendered/.
#
# Использование:
#   ./render-configs.sh
#
# Требования:
#   - .env заполнен (скопирован из .env.example)
#   - envsubst установлен (пакет gettext)
#   - python3 или jq установлен (для очистки _*-ключей из JSON)
#
# Результат:
#   ./rendered/realm-export.json — монтируется в docker-compose в
#   /opt/keycloak/data/import/realm-export.json (--import-realm).
# ==============================================================================

set -euo pipefail

# ==============================================================================
# КОНФИГУРАЦИЯ
# ==============================================================================
SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
ENV_FILE="${SCRIPT_DIR}/.env"
KEYCLOAK_DIR="${SCRIPT_DIR}/keycloak"
RENDERED_DIR="${SCRIPT_DIR}/rendered"

# Шаблон репрезентации и целевой отрендеренный файл
REALM_TEMPLATE="${KEYCLOAK_DIR}/realm-export.json"
REALM_RENDERED="${RENDERED_DIR}/realm-export.json"

# Переменные, которые разрешено подставлять через envsubst.
# Явный список защищает от случайного раскрытия прочих переменных окружения
# и сохраняет валидность JSON (подстановка идёт только для указанных имён).
ENVSUBST_VARS='${OAUTH2_PROXY_CLIENT_SECRET}'

# Цвета для вывода
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

# ==============================================================================
# ЛОГИРОВАНИЕ
# ==============================================================================
log_info()    { echo -e "${GREEN}[INFO]${NC} $1"; }
log_warn()    { echo -e "${YELLOW}[WARN]${NC} $1"; }
log_error()   { echo -e "${RED}[ERROR]${NC} $1" >&2; }

# ==============================================================================
# ПРОВЕРКИ
# ==============================================================================

# Проверка наличия .env
require_env() {
    if [[ ! -f "$ENV_FILE" ]]; then
        log_error "Файл .env не найден: ${ENV_FILE}"
        log_error "Скопируйте .env.example в .env и заполните значения:"
        log_error "  cp .env.example .env && nano .env"
        exit 1
    fi
    log_info "Файл .env найден: ${ENV_FILE}"
}

# Проверка наличия envsubst
require_envsubst() {
    if ! command -v envsubst &>/dev/null; then
        log_error "envsubst не найден."
        log_error "Установите пакет gettext:"
        log_error "  Debian/Ubuntu: sudo apt-get install gettext-base"
        log_error "  RHEL/CentOS:   sudo yum install gettext"
        exit 1
    fi
    log_info "envsubst найден: $(command -v envsubst)"
}

# Определение инструмента для очистки _*-ключей из JSON.
# Предпочтение: python3 → jq. Возвращает имя инструмента или завершает работу.
detect_json_tool() {
    if command -v python3 &>/dev/null; then
        echo "python3"
    elif command -v jq &>/dev/null; then
        echo "jq"
    else
        log_error "Не найден инструмент для обработки JSON (python3 или jq)."
        log_error "Установите один из них:"
        log_error "  python3: sudo apt-get install python3"
        log_error "  jq:      sudo apt-get install jq"
        exit 1
    fi
}

# Проверка наличия шаблона репрезентации
require_realm_template() {
    if [[ ! -f "$REALM_TEMPLATE" ]]; then
        log_error "Шаблон репрезентации не найден: ${REALM_TEMPLATE}"
        exit 1
    fi
    log_info "Шаблон репрезентации найден: ${REALM_TEMPLATE}"
}

# Проверка наличия обязательных переменных в .env
require_vars() {
    # Загружаем .env в окружение для проверок и подстановки
    set -a
    # shellcheck disable=SC1090
    source "$ENV_FILE"
    set +a

    if [[ -z "${OAUTH2_PROXY_CLIENT_SECRET:-}" ]]; then
        log_error "Переменная OAUTH2_PROXY_CLIENT_SECRET не задана в .env"
        exit 1
    fi

    if [[ "${OAUTH2_PROXY_CLIENT_SECRET}" == *CHANGE_ME* ]]; then
        log_warn "OAUTH2_PROXY_CLIENT_SECRET содержит плейсхолдер 'CHANGE_ME'."
        log_warn "Получите реальный секрет клиента в Keycloak и обновите .env."
    fi
}

# ==============================================================================
# РЕНДЕРИНГ РЕПРЕЗЕНТАЦИИ
# ==============================================================================
render_realm() {
    local json_tool
    json_tool="$(detect_json_tool)"
    log_info "Инструмент для очистки JSON: ${json_tool}"

    mkdir -p "$RENDERED_DIR"
    log_info "Директория вывода: ${RENDERED_DIR}"

    # --------------------------------------------------------------------------
    # Шаг 1. Подстановка переменных окружения.
    # Загружаем .env в окружение (если ещё не загружено) и подставляем только
    # переменные из явного списка — это защищает прочие переменные окружения.
    # --------------------------------------------------------------------------
    log_info "Подстановка переменных окружения (envsubst)..."
    set -a
    # shellcheck disable=SC1090
    source "$ENV_FILE"
    set +a

    local substituted
    substituted="$(envsubst "${ENVSUBST_VARS}" < "$REALM_TEMPLATE")"

    # --------------------------------------------------------------------------
    # Шаг 2. Очистка служебных ключей с префиксом "_" из репрезентации.
    # Удаляются на всех уровнях вложенности (например, _ldap_template,
    # _comment_ldap_federation), чтобы импорт в Keycloak не падал на
    # неизвестных полях.
    # --------------------------------------------------------------------------
    log_info "Очистка служебных _*-ключей (${json_tool})..."
    if [[ "$json_tool" == "python3" ]]; then
        printf '%s' "$substituted" | python3 -c '
import json, sys

def clean(obj):
    if isinstance(obj, dict):
        return {
            k: clean(v)
            for k, v in obj.items()
            if not k.startswith("_")
        }
    if isinstance(obj, list):
        return [clean(item) for item in obj]
    return obj

data = json.load(sys.stdin)
print(json.dumps(clean(data), ensure_ascii=False, indent=2))
' > "$REALM_RENDERED"
    else
        # jq: рекурсивно удаляем все ключи, начинающиеся с "_".
        # Функция walk доступна в jq 1.6+.
        printf '%s' "$substituted" | jq 'walk(
            if type == "object"
            then with_entries(select(.key | startswith("_") | not))
            else .
            end
        )' > "$REALM_RENDERED"
    fi

    # --------------------------------------------------------------------------
    # Шаг 3. Валидация результата: результат должен быть валидным JSON.
    # --------------------------------------------------------------------------
    log_info "Валидация результата..."
    if [[ "$json_tool" == "python3" ]]; then
        python3 -c 'import json,sys; json.load(open(sys.argv[1]))' "$REALM_RENDERED"
    else
        jq empty "$REALM_RENDERED"
    fi

    log_info "Отрендерен: ${REALM_TEMPLATE} -> ${REALM_RENDERED}"
}

import os
import re
import secrets
import base64
import urllib.parse
from pathlib import Path

# Путь к вашему файлу (по умолчанию ищет .env или .env.example)
INPUT_FILE = Path(".env")
BACKUP_FILE = Path(".env.bak")

# Алфавиты для генерации
ALPHABET_32_PASS = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789_+=-"
ALPHABET_ALNUM = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789"
ALPHABET_UPPER_ALNUM = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"

def gen_32_pass() -> str:
    """Генерирует 32-символьный пароль со спецсимволами (+, =, _, -)."""
    return "".join(secrets.choice(ALPHABET_32_PASS) for _ in range(32))

def gen_alnum(length: int, upper_only: bool = False) -> str:
    """Генерирует буквенно-цифровой ключ заданной длины."""
    alphabet = ALPHABET_UPPER_ALNUM if upper_only else ALPHABET_ALNUM
    return "".join(secrets.choice(alphabet) for _ in range(length))

def gen_base64(num_bytes: int) -> str:
    """Аналог openssl rand -base64 <num_bytes>."""
    return base64.b64encode(secrets.token_bytes(num_bytes)).decode("ascii")

def rotate_env_file():
    if not INPUT_FILE.exists():
        print(f"Ошибка: Файл {INPUT_FILE} не найден. Укажите правильный путь в INPUT_FILE.")
        return

    # 1. Генерируем новый набор согласованных секретов
    new_secrets = {
        # Стандартные 32-символьные пароли
        "POSTGRES_AUTH_PASSWORD": gen_32_pass(),
        "KEYCLOAK_ADMIN_PASSWORD": gen_32_pass(),
        "KEYCLOAK_DB_PASSWORD": gen_32_pass(),
        "OAUTH2_PROXY_CLIENT_SECRET": gen_32_pass(),
        "REDIS_LANGFUSE_PASSWORD": gen_32_pass(),
        "REDIS_LANGFUSE_ACL_PASSWORD": gen_32_pass(),
        "CLICKHOUSE_PASSWORD": gen_32_pass(),
        "LANGFUSE_DB_PASSWORD": gen_32_pass(),
        "REDIS_FACTORY_ACL_PASSWORD": gen_32_pass(),
        "GATEWAY_DB_PASSWORD": gen_32_pass(),

        # Криптографические соли и ключи сессий
        "LANGFUSE_NEXTAUTH_SECRET": gen_base64(64),
        "LANGFUSE_SALT": gen_base64(32),
        "OAUTH2_PROXY_COOKIE_SECRET": gen_alnum(32), # Ровно 32 байта для AES-256

        # Ключи объектного хранилища S3 (SeaweedFS)
        "S3_ACCESS_KEY": gen_alnum(20, upper_only=True),
        "S3_SECRET_KEY": gen_alnum(40),
    }

    # 2. Автоматически формируем URL-encoded пары (заменяя +, = и т.д.)
    urlencoded_mapping = {
        "CLICKHOUSE_PASSWORD_URLENCODED": "CLICKHOUSE_PASSWORD",
        "REDIS_LANGFUSE_ACL_PASSWORD_URLENCODED": "REDIS_LANGFUSE_ACL_PASSWORD",
        "LANGFUSE_DB_PASSWORD_URLENCODED": "LANGFUSE_DB_PASSWORD",
        "REDIS_FACTORY_ACL_PASSWORD_URLENCODED": "REDIS_FACTORY_ACL_PASSWORD",
        "GATEWAY_DB_PASSWORD_URLENCODED": "GATEWAY_DB_PASSWORD",
    }

    for encoded_key, source_key in urlencoded_mapping.items():
        raw_val = new_secrets[source_key]
        new_secrets[encoded_key] = urllib.parse.quote(raw_val, safe="")

    # 3. Читаем исходный файл и делаем резервную копию
    original_content = INPUT_FILE.read_text(encoding="utf-8")
    BACKUP_FILE.write_text(original_content, encoding="utf-8")

    updated_lines = []
    replaced_count = 0

    # Регулярное выражение для поиска строк вида KEY='value' или KEY=value
    pattern = re.compile(r"^([A-Z0-9_]+)=(.*)$")

    for line in original_content.splitlines():
        match = pattern.match(line.strip())
        if match:
            key = match.group(1)
            if key in new_secrets:
                # Сохраняем формат с одинарными кавычками, как в оригинале
                updated_lines.append(f"{key}='{new_secrets[key]}'")
                replaced_count += 1
                continue
        updated_lines.append(line)

    # 4. Записываем обновленный файл
    INPUT_FILE.write_text("\n".join(updated_lines) + "\n", encoding="utf-8")

    print(f"Успешно обновлено секретов: {replaced_count}")
    print(f"Старая версия сохранена в: {BACKUP_FILE.resolve()}")
    print("ВАЖНО: Удалите .env.bak после применения изменений в инфраструктуре!")

if __name__ == "__main__":
    rotate_env_file()

@asynccontextmanager
async def lifespan(app: FastAPI) -> AsyncGenerator[None, None]:
    """Startup / shutdown."""
    logger.info("Starting AG-UI Gateway", environment=settings.environment)

    # Redis connection pool
    # redis:// or rediss:// selects plaintext/TLS. Do not pass redis-py ssl kwargs,
    # which are incompatible with the pinned Redis 8 client.
    app.state.redis = redis.from_url(
        settings.redis_factory_url,
        decode_responses=True,
    )

    # Проверка подключения к Redis
    try:
        await app.state.redis.ping()
        logger.info("Redis connection established")
    except Exception as e:
        logger.error("Failed to connect to Redis", error=str(e))
        raise

    # PostgreSQL connection pool (для реестра ранов)
    # Blueprint §3.1: autocommit=True, row_factory=dict_row, prepare_threshold=0
    # TLS включён через схему postgresql+ssl:// и sslmode=require в URL
    app.state.pg_pool = AsyncConnectionPool(
        conninfo=settings.postgres_runtime_url,
        min_size=2,
        max_size=10,

@app.post("/api/agent")
async def run_agent(input: RunAgentInput, request: Request):
    """
    Приём RunAgentInput, постановка задачи в Redis через Taskiq, SSE-relay событий.

    Это основной endpoint AG-UI протокола:
      1. Клиент отправляет POST с RunAgentInput
      2. Gateway создаёт запись в реестре ранов (PostgreSQL) со статусом QUEUED
      3. Gateway ставит задачу в Redis через Taskiq broker
      4. Gateway открывает SSE-соединение и читает события из Redis Streams
      5. События стримятся клиенту в формате AG-UI
    """
    user = request.headers.get("X-Forwarded-User", "anonymous")

    logger.info(
        "RunAgentInput received",
        thread_id=input.thread_id,
        run_id=input.run_id,
        user=user,
    )

    # Langfuse trace
    trace = app.state.langfuse.trace(
        name="gateway.run_agent",
        metadata={
            "thread_id": input.thread_id,
            "run_id": input.run_id,
            "user": user,
        },
    )

    # 1. Создание записи в реестре ранов (PostgreSQL)
    try:
        async with app.state.pg_pool.connection() as conn:
            async with conn.cursor() as cur:
                await cur.execute(
                    """
                    INSERT INTO runs (run_id, thread_id, status, user, payload, created_at, updated_at)
                    VALUES (%s, %s, %s, %s, %s, %s, %s)
                    ON CONFLICT (run_id) DO NOTHING
                    """,
                    (
                        input.run_id,
                        input.thread_id,
                        "QUEUED",
                        user,
                        orjson.dumps({
                            "messages": input.messages,
                            "state": input.state,
                            "tools": input.tools,
                            "context": input.context,
                            "forwarded_props": input.forwarded_props,
                        }).decode("utf-8"),
                        datetime.utcnow(),
                        datetime.utcnow(),
                    ),
                )
        logger.info("Run created in registry", run_id=input.run_id, status="QUEUED")
    except Exception as e:
        logger.error("Failed to create run in registry", run_id=input.run_id, error=str(e))
        raise HTTPException(status_code=500, detail="Failed to create run")

    # 2. Постановка задачи в Redis через Taskiq broker
    task_payload = {
        "run_id": input.run_id,
        "thread_id": input.thread_id,
        "user": user,
        "messages": input.messages,
        "state": input.state,
        "tools": input.tools,
        "context": input.context,
        "forwarded_props": input.forwarded_props,
    }

    try:
        # Отправка задачи через Taskiq broker
        await app.state.broker.send_task(
            task_name=TASK_RUN_AGENT,
            args=[task_payload],
            queue=TASK_QUEUE_KEY,
        )
        logger.info("Task enqueued via Taskiq", run_id=input.run_id)
    except Exception as e:
        logger.error("Failed to enqueue task", run_id=input.run_id, error=str(e))
        # Обновляем статус рана на FAILED
        async with app.state.pg_pool.connection() as conn:
            async with conn.cursor() as cur:
                await cur.execute(
                    """
                    UPDATE runs
                    SET status = %s, updated_at = %s
                    WHERE run_id = %s
                    """,
                    ("FAILED", datetime.utcnow(), input.run_id),
                )
        raise HTTPException(status_code=500, detail="Failed to enqueue task")

    # 3. SSE-relay: читаем события из Redis Streams и стримим клиенту
    async def event_generator() -> AsyncGenerator[str, None]:
        stream_key = _stream_key(input.thread_id)
        # Начинаем с последнего сообщения стрима, чтобы не читать историю прошлых ранов
        last_id = "$"

        try:
            while True:
                # XREAD BLOCK: ждём новые события
                result = await app.state.redis.xread(
                    {stream_key: last_id},
                    count=10,
                    block=5000,  # 5 секунд
                )

                if result:
                    for _, messages in result:
                        for msg_id, msg_data in messages:
                            last_id = msg_id

                            # Формируем SSE-событие
                            event_type = msg_data.get("type", "custom")
                            event_data = msg_data.get("data", "{}")

                            yield (
                                f"id: {msg_id}\n"
                                f"event: {event_type}\n"
                                f"data: {event_data}\n\n"
                            )

                            # Если это терминальное событие — закрываем стрим
                            if event_type in ("RUN_FINISHED", "RUN_ERROR"):
                                logger.info(
                                    "Run finished",
                                    run_id=input.run_id,
                                    event_type=event_type,
                                )
                                return

                # Heartbeat (чтобы nginx не закрывал соединение)
                yield ": heartbeat\n\n"
                await asyncio.sleep(settings.sse_heartbeat_interval)

        except asyncio.CancelledError:
            logger.info("SSE connection closed by client", run_id=input.run_id)
        except Exception as e:
            logger.error("SSE error", run_id=input.run_id, error=str(e))
            yield f"event: error\ndata: {str(e)}\n\n"

    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no",
        },
    )


@app.get("/api/threads/{thread_id}/events")
async def reconnect_events(
        thread_id: str,
        request: Request,
        last_event_id: Optional[str] = None,
):
    """
    Реконнект SSE-соединения по thread_id с Last-Event-ID.
    Используется, когда клиент потерял соединение и хочет продолжить чтение событий.
    Заголовок `Last-Event-ID` — стандартный SSE-механизм реконнекта.
    """
    user = request.headers.get("X-Forwarded-User", "anonymous")

    # Читаем заголовок `Last-Event-ID` (стандартный механизм)
    # Если заголовка нет — используем query-параметр (для обратной совместимости)
    header_last_id = request.headers.get("Last-Event-ID")
    if header_last_id:
        start_id = header_last_id
    else:
        start_id = last_event_id if last_event_id else "0-0"

    logger.info(
        "Reconnect requested",
        thread_id=thread_id,
        last_event_id=start_id,
        user=user,
    )

    stream_key = _stream_key(thread_id)

    async def event_generator() -> AsyncGenerator[str, None]:
        try:
            # Читаем пропущенные события (с последнего известного)
            result = await app.state.redis.xread({stream_key: start_id}, count=100)

            if result:
                for _, messages in result:
                    for msg_id, msg_data in messages:
                        event_type = msg_data.get("type", "custom")
                        event_data = msg_data.get("data", "{}")

                        yield (
                            f"id: {msg_id}\n"
                            f"event: {event_type}\n"
                            f"data: {event_data}\n\n"
                        )

            # Продолжаем live-tail
            last_id = start_id
            while True:
                result = await app.state.redis.xread(
                    {stream_key: last_id}, count=10, block=5000
                )

                if result:
                    for _, messages in result:
                        for msg_id, msg_data in messages:
                            last_id = msg_id
                            event_type = msg_data.get("type", "custom")
                            event_data = msg_data.get("data", "{}")

                            yield (
                                f"id: {msg_id}\n"
                                f"event: {event_type}\n"
                                f"data: {event_data}\n\n"
                            )

                            if event_type in ("RUN_FINISHED", "RUN_ERROR"):
                                return

                yield ": heartbeat\n\n"
                await asyncio.sleep(settings.sse_heartbeat_interval)

        except asyncio.CancelledError:
            logger.info("Reconnect SSE closed", thread_id=thread_id)
        except Exception as e:
            logger.error("Reconnect SSE error", thread_id=thread_id, error=str(e))
            yield f"event: error\ndata: {str(e)}\n\n"

    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no",
        },
    )


@app.post("/api/threads/{thread_id}/resume")
async def resume_run(
        thread_id: str, resume_req: ResumeRequest, request: Request
):
    """
    Resume HITL-паузы с идемпотентностью.

    Воркер остановился на interrupt(), ждёт решения человека.
    Gateway:
      1. Проверяет идемпотентность по interrupt_id (нет ли уже обработанного)
      2. Обновляет статус рана на WAITING_HUMAN
      3. Ставит задачу resume в очередь через Taskiq
      4. Воркер продолжает выполнение
    """
    user = request.headers.get("X-Forwarded-User", "anonymous")

    logger.info(
        "Resume requested",
        thread_id=thread_id,
        interrupt_id=resume_req.interrupt_id,
        user=user,
    )

    # Langfuse trace
    trace = app.state.langfuse.trace(
        name="gateway.resume_run",
        metadata={
            "thread_id": thread_id,
            "interrupt_id": resume_req.interrupt_id,
            "user": user,
        },
    )

    # 1. Идемпотентность: проверяем, нет ли уже обработанного interrupt_id
    try:
        async with app.state.pg_pool.connection() as conn:
            async with conn.cursor() as cur:
                # Проверяем, есть ли ран с таким interrupt_id и статусом WAITING_HUMAN
                await cur.execute(
                    """
                    SELECT run_id, status FROM runs
                    WHERE thread_id = %s AND interrupt_id = %s
                    """,
                    (thread_id, resume_req.interrupt_id),
                )
                existing = await cur.fetchone()

                if existing:
                    existing_run_id, existing_status = existing

                    # Если уже обработан (SUCCEEDED/FAILED/CANCELLED) — возвращаем 409
                    if existing_status in ("SUCCEEDED", "FAILED", "CANCELLED"):
                        logger.warning(
                            "Resume rejected: interrupt already processed",
                            thread_id=thread_id,
                            interrupt_id=resume_req.interrupt_id,
                            existing_status=existing_status,
                        )
                        raise HTTPException(
                            status_code=409,
                            detail=f"Interrupt already processed: {existing_status}",
                        )

                    # Если уже WAITING_HUMAN — идемпотентно возвращаем 200
                    if existing_status == "WAITING_HUMAN":
                        logger.info(
                            "Resume idempotent: already waiting",
                            thread_id=thread_id,
                            interrupt_id=resume_req.interrupt_id,
                        )
                        return JSONResponse(
                            {"status": "already_waiting", "thread_id": thread_id}
                        )

                # 2. Обновляем статус рана на WAITING_HUMAN
                await cur.execute(
                    """
                    UPDATE runs
                    SET status = %s, interrupt_id = %s, updated_at = %s
                    WHERE thread_id = %s AND status IN ('RUNNING', 'WAITING_HUMAN')
                    """,
                    ("WAITING_HUMAN", resume_req.interrupt_id, datetime.utcnow(), thread_id),
                )

                if cur.rowcount == 0:
                    logger.error(
                        "Resume failed: no running run found",
                        thread_id=thread_id,
                    )
                    raise HTTPException(
                        status_code=404,
                        detail="No running run found for this thread",
                    )
    except HTTPException:
        raise
    except Exception as e:
        logger.error("Failed to update run status", thread_id=thread_id, error=str(e))
        raise HTTPException(status_code=500, detail="Failed to update run status")

    # 3. Постановка задачи resume в очередь через Taskiq
    resume_payload = {
        "type": "resume",
        "thread_id": thread_id,
        "interrupt_id": resume_req.interrupt_id,
        "value": resume_req.value,
        "user": user,
    }

    try:
        await app.state.broker.send_task(
            task_name=TASK_RESUME_RUN,
            args=[resume_payload],
            queue=TASK_QUEUE_KEY,
        )
        logger.info(
            "Resume task enqueued via Taskiq",
            thread_id=thread_id,
            interrupt_id=resume_req.interrupt_id,
        )
    except Exception as e:
        logger.error("Failed to enqueue resume task", thread_id=thread_id, error=str(e))
        raise HTTPException(status_code=500, detail="Failed to enqueue resume task")

    return JSONResponse({"status": "accepted", "thread_id": thread_id})

#!/usr/bin/env bash
# Идемпотентная инициализация БД Keycloak и Langfuse для ВМ №1.
set -Eeuo pipefail

: "${POSTGRES_USER:?POSTGRES_USER is required}"
: "${POSTGRES_DB:?POSTGRES_DB is required}"
: "${KEYCLOAK_DB_USER:?KEYCLOAK_DB_USER is required}"
: "${KEYCLOAK_DB_PASSWORD:?KEYCLOAK_DB_PASSWORD is required}"
: "${LANGFUSE_DB_USER:?LANGFUSE_DB_USER is required}"
: "${LANGFUSE_DB_PASSWORD:?LANGFUSE_DB_PASSWORD is required}"

create_role() {
    local role="$1"
    local password="$2"

    psql -v ON_ERROR_STOP=1 \
        --username "$POSTGRES_USER" \
        --dbname "$POSTGRES_DB" \
        --set=role_name="$role" \
        --set=role_password="$password" <<'EOSQL'
SELECT format('CREATE ROLE %I LOGIN', :'role_name')
WHERE NOT EXISTS (SELECT FROM pg_roles WHERE rolname = :'role_name')
\gexec
SELECT format('ALTER ROLE %I WITH LOGIN ENCRYPTED PASSWORD %L', :'role_name', :'role_password')
\gexec
EOSQL
}

create_database() {
    local database="$1"
    local owner="$2"

    psql -v ON_ERROR_STOP=1 \
        --username "$POSTGRES_USER" \
        --dbname "$POSTGRES_DB" \
        --set=db_name="$database" \
        --set=db_owner="$owner" <<'EOSQL'
SELECT format('CREATE DATABASE %I OWNER %I', :'db_name', :'db_owner')
WHERE NOT EXISTS (SELECT FROM pg_database WHERE datname = :'db_name')
\gexec
SELECT format('ALTER DATABASE %I OWNER TO %I', :'db_name', :'db_owner')
\gexec
SELECT format('GRANT ALL PRIVILEGES ON DATABASE %I TO %I', :'db_name', :'db_owner')
\gexec
EOSQL

    psql -v ON_ERROR_STOP=1 \
        --username "$POSTGRES_USER" \
        --dbname "$database" \
        --set=db_owner="$owner" <<'EOSQL'
GRANT ALL ON SCHEMA public TO :"db_owner";
ALTER SCHEMA public OWNER TO :"db_owner";
EOSQL
}

echo 'Creating/updating Keycloak and Langfuse database roles.'
create_role "$KEYCLOAK_DB_USER" "$KEYCLOAK_DB_PASSWORD"
create_role "$LANGFUSE_DB_USER" "$LANGFUSE_DB_PASSWORD"

create_database keycloak "$KEYCLOAK_DB_USER"
create_database langfuse "$LANGFUSE_DB_USER"

echo 'Database initialization completed.'
