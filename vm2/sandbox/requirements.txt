# ==============================================================================
# requirements.txt — Базовые зависимости для песочниц gVisor (ВМ №2)
# Production Blueprint v5.2
# ==============================================================================
# Назначение: Минимальный набор библиотек для обработки документов и данных
# в изолированных песочницах. Используется как дефолтный TOOLSET_REQS в
# Dockerfile.toolset.
#
# ВАЖНО:
#   - Все пакеты должны быть предварительно загружены в wheel-cache на ВМ №2.
#   - Песочницы не имеют сетевого доступа (PIP_NO_INDEX=1, PIP_FIND_LINKS=/app/wheel-cache).
#   - Версии зафиксированы (==) для воспроизводимости сборок.
#   - Для специфичных toolset (toolset-pdf, toolset-excel) создаются отдельные
#     requirements-файлы (requirements-pdf.txt, requirements-excel.txt).
# ==============================================================================

# ==============================================================================
# ОБРАБОТКА ТАБЛИЧНЫХ ДАННЫХ
# ==============================================================================
# pandas — основной инструмент для работы с табличными данными (Excel, CSV)
# Используется для: сверки позиций актов, формирования выходных шаблонов SAP/Визирь
pandas==2.1.4

# openpyxl — чтение/запись Excel-файлов (.xlsx)
# Используется для: шаблонов SAP (Z_GFL_REGFILSRV), реестров ставок
openpyxl==3.1.2

# xlrd — чтение старых Excel-файлов (.xls)
# Используется для: совместимости со старыми реестрами подрядчиков
xlrd==2.0.1

# ==============================================================================
# РАБОТА С PDF
# ==============================================================================
# PyPDF2 — базовая работа с PDF (чтение метаданных, извлечение текста)
# Используется для: первичного анализа документов, извлечения реквизитов
PyPDF2==3.0.1

# pdfplumber — точное извлечение таблиц и текста из PDF с координатами
# Используется для: извлечения позиций из КС-2, КС-3, УПД
pdfplumber==0.10.3

# ==============================================================================
# ОБРАБОТКА ДАННЫХ
# ==============================================================================
# python-dateutil — парсинг и манипуляция датами
# Используется для: работы с периодами действия ставок, датами договоров
python-dateutil==2.8.2

# jsonschema — валидация JSON-структур
# Используется для: проверки входных/выходных данных инструментов
jsonschema==4.20.0

# ==============================================================================
# УТИЛИТЫ
# ==============================================================================
# chardet — определение кодировки текстовых файлов
# Используется для: обработки CSV и текстовых файлов от подрядчиков
chardet==5.2.0

# ==============================================================================
# ПРИМЕЧАНИЯ ПО ИСПОЛЬЗОВАНИЮ
# ==============================================================================
# 1. Этот файл используется по умолчанию в Dockerfile.toolset:
#    docker build -t factory/toolset-base:1.0.0 -f sandbox/Dockerfile.toolset .
#
# 2. Для специфичных toolset создаются отдельные файлы:
#    - requirements-pdf.txt (для toolset-pdf: добавляет pdfplumber, PyMuPDF)
#    - requirements-excel.txt (для toolset-excel: добавляет xlsxwriter, openpyxl)
#    - requirements-calc.txt (для toolset-calc: добавляет numpy, scipy)
#
# 3. Сборка специфичного toolset:
#    docker build --build-arg TOOLSET_REQS=requirements-excel.txt \
#      -t factory/toolset-excel:1.0.0 -f sandbox/Dockerfile.toolset .
#
# 4. Все пакеты должны быть в wheel-cache на ВМ №2:
#    /app/wheel-cache/pandas-2.1.4-cp313-cp313-manylinux_2_17_x86_64.manylinux2014_x86_64.whl
#    /app/wheel-cache/openpyxl-3.1.2-py2.py3-none-any.whl
#    ...
#
# 5. Если пакета нет в кэше, сборка упадёт с ошибкой:
#    ERROR: Could not find a version that satisfies the requirement pandas==2.1.4
