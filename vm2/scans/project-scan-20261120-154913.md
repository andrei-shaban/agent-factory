#!/usr/bin/env bash
# ==============================================================================
# backup.sh — Автоматический бэкап PostgreSQL Runtime (ВМ №2)
# Production Blueprint v5.2
# ==============================================================================
# Назначение:
#   1. Создание full backup через pg_basebackup в SeaweedFS S3 (бакет postgres-backups).
#   2. Архивация WAL-файлов через pg_archivecleanup (ретентивность 30 дней).
#   3. Логирование результатов и отправка событий в Redis (опционально).
#
# Запуск:
#   - Вручную: sudo ./backups/backup.sh
#   - По cron: 0 2 * * * /opt/factory-vm2/backups/backup.sh >> /var/log/factory-backup.log 2>&1
#
# Требования:
#   - Запуск от пользователя postgres или root с доступом к контейнеру vm2-postgres-runtime
#   - Docker Compose запущен
#   - SeaweedFS S3 доступен
# ==============================================================================

set -euo pipefail

# ==============================================================================
# КОНФИГУРАЦИЯ
# ==============================================================================
SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
PROJECT_DIR="$(dirname "$SCRIPT_DIR")"
ENV_FILE="${PROJECT_DIR}/.env"
LOG_FILE="/var/log/factory-vm2-backup.log"
BACKUP_RETENTION_DAYS=30

# Загрузка переменных из .env
if [[ -f "$ENV_FILE" ]]; then
    set -a
    source "$ENV_FILE"
    set +a
fi

# Параметры подключения к PostgreSQL Runtime
PG_CONTAINER="vm2-postgres-runtime"
PG_USER="${POSTGRES_RUNTIME_USER:-factory_runtime}"

# Цвета для вывода
RED='\033[0;31m'

YELLOW='\033[1;33m'

# ==============================================================================

log_info() {

}

require_env() {

        log_error "Файл .env не найден: $ENV_FILE"

        exit 1
    fi
}

require_docker() {

# ==============================================================================
# 1. FULL BACKUP (pg_basebackup)
# ==============================================================================
create_full_backup() {
    log_info "=== Шаг 1: Создание full backup через pg_basebackup ==="

    local backup_date
    backup_date=$(date +%Y%m%d_%H%M%S)
    local backup_dir="/tmp/pg_backup_${backup_date}"
    local s3_key="full/${backup_date}"

    log_info "Создание бэкапа в ${backup_dir}..."

    # Выполняем pg_basebackup внутри контейнера
    docker exec "$PG_CONTAINER" bash -c "
        pg_basebackup \
            -U ${PG_BACKUP_USER} \
            -D ${backup_dir} \
            -Ft \
            -z \
            -P \
            --checkpoint=fast \
            --wal-method=stream
    " 2>&1 | tee -a "$LOG_FILE"

    if [[ $? -ne 0 ]]; then
        log_error "pg_basebackup завершился с ошибкой"
        rm -rf "$backup_dir"
        return 1
    fi

    log_success "Бэкап создан: ${backup_dir}"

    # Загрузка в S3
    log_info "Загрузка бэкапа в S3: ${S3_BUCKET}/${s3_key}..."

    # Архивируем и загружаем
    tar -czf "/tmp/pg_backup_${backup_date}.tar.gz" -C /tmp "pg_backup_${backup_date}"

    # Используем curl для загрузки в S3 (SeaweedFS совместим с AWS S3 API)
    local file_size
    file_size=$(stat -c%s "/tmp/pg_backup_${backup_date}.tar.gz")

    curl -s -X PUT "${S3_ENDPOINT}/${S3_BUCKET}/${s3_key}.tar.gz" \
        -H "Content-Type: application/gzip" \
        -H "Authorization: AWS ${S3_ACCESS_KEY}:${S3_SECRET_KEY}" \
        --data-binary "@/tmp/pg_backup_${backup_date}.tar.gz" \
        2>&1 | tee -a "$LOG_FILE"

    if [[ $? -eq 0 ]]; then
        log_success "Бэкап загружен в S3: ${s3_key}.tar.gz (${file_size} байт)"
    else
        log_error "Не удалось загрузить бэкап в S3"
        rm -f "/tmp/pg_backup_${backup_date}.tar.gz"
        return 1
    fi

    # Очистка временных файлов
    rm -rf "$backup_dir"
    rm -f "/tmp/pg_backup_${backup_date}.tar.gz"

    log_success "Full backup завершён"
}

# ==============================================================================
# 01-create-databases.sql.sh — ВМ №2: инициализация БД factory, пользователей, базовых таблиц
# Production Blueprint v5.2
# ==============================================================================
# Выполняется автоматически при ПЕРВОМ запуске контейнера postgres-runtime.
# Формат .sql.sh гарантирует, что официальный образ исполнит скрипт как bash
# с доступом к переменным окружения из docker-compose.
#
# По Blueprint §3.1: ВМ №2 обслуживает чекпоинты LangGraph, реестр ранов
# фабрики и векторный поиск pgvector.
#
# Переменные подставляются из .env → docker-compose → environment контейнера:
#   FACTORY_DB_USER, FACTORY_DB_PASSWORD
#   WORKER_DB_USER, WORKER_DB_PASSWORD
#   GATEWAY_DB_USER, GATEWAY_DB_PASSWORD
# ==============================================================================
set -euo pipefail

echo "=== Создание реестра ранов фабрики (ВМ №2) ==="

# --------------------------------------------------------------------------
# 1. База factory (реестр ранов, чекпоинты)
# --------------------------------------------------------------------------
echo "Создание БД: factory, владелец: ${FACTORY_DB_USER}"
psql -v ON_ERROR_STOP=1 --username "$POSTGRES_USER" --dbname "$POSTGRES_DB" <<-EOSQL
    CREATE DATABASE factory;
    CREATE USER ${FACTORY_DB_USER} WITH ENCRYPTED PASSWORD '${FACTORY_DB_PASSWORD}';
    GRANT ALL PRIVILEGES ON DATABASE factory TO ${FACTORY_DB_USER};
    ALTER DATABASE factory OWNER TO ${FACTORY_DB_USER};
EOSQL

# Настройка схемы внутри базы
psql -v ON_ERROR_STOP=1 --username "$POSTGRES_USER" --dbname "factory" <<-EOSQL
    GRANT ALL ON SCHEMA public TO ${FACTORY_DB_USER};
EOSQL

echo "База factory готова."

# 2. Пользователь Worker (для LangGraph, чекпоинты, все таблицы)
# --------------------------------------------------------------------------
echo "Создание пользователя Worker: ${WORKER_DB_USER}"
psql -v ON_ERROR_STOP=1 --username "$POSTGRES_USER" --dbname "factory" <<-EOSQL
    DO \$\$
    BEGIN
        IF NOT EXISTS (SELECT FROM pg_catalog.pg_roles WHERE rolname = '${WORKER_DB_USER}') THEN
            CREATE ROLE ${WORKER_DB_USER} WITH LOGIN ENCRYPTED PASSWORD '${WORKER_DB_PASSWORD}';
        END IF;
    END
    \$\$;
    GRANT CONNECT ON DATABASE factory TO ${WORKER_DB_USER};
    GRANT USAGE ON SCHEMA public TO ${WORKER_DB_USER};
    GRANT ALL ON ALL TABLES IN SCHEMA public TO ${WORKER_DB_USER};
    GRANT ALL ON ALL SEQUENCES IN SCHEMA public TO ${WORKER_DB_USER};
EOSQL

echo "Пользователь ${WORKER_DB_USER} готов."

# --------------------------------------------------------------------------
# 3. Пользователь Gateway (только таблица runs, read/write)
# --------------------------------------------------------------------------
echo "Создание пользователя Gateway: ${GATEWAY_DB_USER}"
psql -v ON_ERROR_STOP=1 --username "$POSTGRES_USER" --dbname "factory" <<-EOSQL
    DO \$\$
    BEGIN
        IF NOT EXISTS (SELECT FROM pg_catalog.pg_roles WHERE rolname = '${GATEWAY_DB_USER}') THEN
            CREATE ROLE ${GATEWAY_DB_USER} WITH LOGIN ENCRYPTED PASSWORD '${GATEWAY_DB_PASSWORD}';
        END IF;
    END
    \$\$;
    GRANT CONNECT ON DATABASE factory TO ${GATEWAY_DB_USER};
    GRANT USAGE ON SCHEMA public TO ${GATEWAY_DB_USER};

## Примечания по безопасности

- **Air-Gap**: образы зафиксированы по digest, зависимости только из внутреннего зеркала, CDN запрещён, телеметрия отключена.
- **mTLS**: клиентские сертификаты воркера подписаны корпоративным CA; используются при обращении к .228, рабочей станции и ВМ №4. Ключи (`vm2-client.key`) изолированы на ВМ №2, права `0600`, монтируются в контейнер worker read-only.
- **Песочницы**: `runsc` (gVisor), `--network none`, read-only rootfs, tmpfs, `cap-drop ALL`, `no-new-privileges`, лимиты CPU/memory/pids, непривилегированный uid. Секреты внутрь не передаются. Песочницы не имеют сетевого доступа никуда: зависимости доставляются воркером через локальный wheel-cache (read-only том).
- **Docker socket proxy**: whitelist `CONTAINERS=1, POST=1, EXEC=1`. Доступ к `NETWORKS`, `VOLUMES`, `IMAGES`, `PRIVILEGED` запрещён. Прямой docker.sock в контейнеры не передаётся.
- **PostgreSQL Runtime**: отдельный том, жёсткое лимитирование памяти (`shared_buffers`, `work_mem`) для исключения конкуренции с Redis и песочницами. WAL-архивация в SeaweedFS.
- **Redis фабрики**: AOF включён (`appendonly yes`, `appendfsync everysec`), ACL `agui_svc`, жёсткий `MAXLEN ~ N` + TTL на Streams. Размещён локально к воркеру для минимизации round-trip при `XADD` на каждое событие.
- **Часовой пояс**: `TZ=Asia/Yekaterinburg` (Екатеринбург).
- **Кодировка**: все текстовые конфигурации — UTF-8 без BOM, окончания строк LF.

class EventType(str, Enum):
    """Типы событий AG-UI протокола."""
    RUN_STARTED = "RUN_STARTED"
    RUN_FINISHED = "RUN_FINISHED"
    RUN_FAILED = "RUN_FAILED"
    NODE_STARTED = "NODE_STARTED"
    NODE_FINISHED = "NODE_FINISHED"
    HITL_INTERRUPT = "HITL_INTERRUPT"
    HITL_RESUMED = "HITL_RESUMED"
    TOOL_EXECUTED = "TOOL_EXECUTED"
    SSE_MESSAGE = "SSE_MESSAGE"
    ERROR = "ERROR"


class EventBus:
    """
    Шина событий AG-UI.
    Публикует события в Redis Streams для SSE-relay через Gateway.
    """

    def __init__(self, settings: Optional[Settings] = None):
        self.settings = settings or get_settings()
        self._redis: Optional[redis.Redis] = None

    async def _get_redis(self) -> redis.Redis:
        """Ленивая инициализация Redis клиента."""
        if self._redis is None:
            self._redis = redis.from_url(
                self.settings.redis_factory_url,
                decode_responses=True,
            )
            logger.info("EventBus Redis client initialized")
        return self._redis

    async def close(self) -> None:
        """Закрытие Redis соединения."""
        if self._redis:
            await self._redis.close()
            self._redis = None

    async def publish_event(
            self,
            event_type: EventType,
            thread_id: str,
            run_id: str,
            data: Dict[str, Any],
            node_name: Optional[str] = None,
            timestamp: Optional[str] = None,
    ) -> str:
        """
        Публикация события в Redis Streams.

        Args:
            event_type: Тип события (EventType).
            thread_id: Идентификатор треда (ключ стрима: agui:stream:{thread_id}).
            run_id: Идентификатор запуска.
            data: Полезная нагрузка события (словарь).
            node_name: Имя узла графа (для NODE_STARTED/NODE_FINISHED).
            timestamp: Временная метка (ISO 8601, по умолчанию — текущее время).

        Returns:
            ID добавленного сообщения в стрим (например, "1234567890123-0").
        """
        redis_client = await self._get_redis()
        stream_key = f"agui:stream:{thread_id}"

        timestamp = timestamp or datetime.now(timezone.utc).isoformat()

        event = {
            "type": event_type.value,
            "run_id": run_id,
            "thread_id": thread_id,
            "timestamp": timestamp,
            "data": json.dumps(data, ensure_ascii=False, default=str),
        }

        if node_name:
            event["node"] = node_name

        # Публикация в стрим с ограничением размера (MAXLEN ~ 10000)
        # ~ означает approximate trimming (быстрее, чем точное ограничение)
        message_id = await redis_client.xadd(
            name=stream_key,
            fields=event,
            maxlen=10000,
            approximate=True,
        )

        logger.debug(
            "Event published",
            event_type=event_type.value,
            thread_id=thread_id,
            run_id=run_id,
            message_id=message_id,
        )

        return message_id

async def load_and_ocr(state: DocumentState) -> DocumentState:

async def match_rates(state: DocumentState) -> DocumentState:
    """Ф3: Сверка позиций со ставками ЕРР (корп PG, нечёткий поиск)."""
    run_id = state["run_id"]
    thread_id = state["thread_id"]
    node_name = "match_rates"
    log = logger.bind(run_id=run_id, node=node_name)

    await event_bus.publish_node_started(thread_id, run_id, node_name)
    _start_span(state, node_name, input_data={"positions_count": len(state.get("positions", [])), "contract_id": state.get("contract_id")})

    try:
        log.info("Matching positions against rate registry")

        contract_id = state.get("contract_id")
        verification: List[Dict[str, Any]] = []
        mismatches: List[Dict[str, Any]] = []

        for pos in state["positions"]:
            rates = await corp_pg_client.get_rates_for_contract(
                contract_id=contract_id,
                as_of_date=None,
            )
            matched = _fuzzy_match_position(pos["name"], rates)

            if matched is None:
                mismatches.append({
                    "position": pos,
                    "reason": "missing_rate",
                    "confidence": 0.0,
                })
                verification.append({"position_number": pos["number"], "status": "missing_rate"})
            else:
                price_diff = abs(float(pos.get("price", 0)) - float(matched["price"]))
                if price_diff > 0.01:
                    mismatches.append({
                        "position": pos,
                        "expected_price": matched["price"],
                        "actual_price": pos.get("price"),
                        "difference": price_diff,
                        "reason": "mismatched",
                        "confidence": 0.9,
                        "page_number": pos.get("page"),
                    })
                    verification.append({"position_number": pos["number"], "status": "mismatched"})
                else:
                    verification.append({"position_number": pos["number"], "status": "matched"})

        log.info("Matching complete", verified=len(verification), mismatches=len(mismatches))

        _end_span(state, node_name, output_data={"verified": len(verification), "mismatches": len(mismatches)})
        await event_bus.publish_node_finished(thread_id, run_id, node_name, output={"verified": len(verification), "mismatches": len(mismatches)})

        return {
            "status": "rates_matched",
            "verification": verification,
            "mismatches": mismatches,
            "messages": [{"role": "assistant", "content": f"Сверка: расхождений {len(mismatches)}"}],
        }
    except Exception as e:
        log.error("Rate matching failed", error=str(e))
        _record_error(state, node_name, e)
        await event_bus.publish_error(thread_id, run_id, e, node_name=node_name)
        raise


def _fuzzy_match_position(name: str, rates: List[Dict[str, Any]]) -> Optional[Dict[str, Any]]:
    """Нечёткое сопоставление наименования позиции со ставками ЕРР."""
    from difflib import SequenceMatcher
    best, best_score = None, 0.0
    for r in rates:
        score = SequenceMatcher(None, name.lower(), r["item_name"].lower()).ratio()
        if score > best_score:
            best, best_score = r, score
    return best if best_score >= 0.75 else None


async def specialist_review(state: DocumentState) -> DocumentState:

            "form_id": "specialist_verification",
            "title": "Проверка расхождений",
            "mismatches": state["mismatches"],
            "allowed_actions": ["approve", "correct", "reject"],

def _apply_specialist_corrections(
    positions: List[Dict[str, Any]], decision: Dict[str, Any]
) -> List[Dict[str, Any]]:
    """Применяет корректировки специалиста к позициям."""
    if not decision.get("corrections"):
        return positions
    corrected = [dict(p) for p in positions]
    for corr in decision["corrections"]:
        for p in corrected:
            if p["number"] == corr["position_number"]:
                p.update(corr["values"])
    return corrected

Назначение:

from pathlib import Path

# Regex для парсинга PEP 503 Simple API (строго определённый формат, bs4 не нужен)
_SIMPLE_API_LINK_RE = re.compile(r'<a\s+[^>]*href="([^"]+)"[^>]*>([^<]+)</a>', re.IGNORECASE)


class PyPIError(Exception):

    async def get_package_urls(self, package_name: str, version: Optional[str] = None) -> List[Dict[str, str]]:
        """
        Получение списка URL для скачивания пакета через PEP 503 Simple API.

        Args:
            package_name: Имя пакета (например, 'pandas').
            version: Требуемая версия (например, '2.1.4'). Если None, вернёт все.

        Returns:
            Список словарей {"url": "...", "filename": "...", "requires-python": "..."}
        """
        client = self._get_client()
        # PEP 503: имена пакетов нормализуются (lowercase, дефисы/точки/подчёркивания -> дефис)
        normalized_name = re.sub(r"[-_.]+", "-", package_name).lower()

        logger.debug("Querying Simple API", package=package_name, normalized=normalized_name)

        try:
            resp = await client.get(f"/simple/{normalized_name}/", headers={"Accept": "text/html"})
            resp.raise_for_status()
        except httpx.HTTPStatusError as e:
            if e.response.status_code == 404:
                raise PyPIError(f"Package '{package_name}' not found in mirror")
            raise PyPIError(f"Simple API error for '{package_name}': {e}")
        except httpx.TransportError as e:
            raise PyPIError(f"Mirror unreachable: {e}")

        # Парсинг HTML (без внешних зависимостей)
        links = []
        for match in _SIMPLE_API_LINK_RE.finditer(resp.text):
            url, filename = match.groups()
            # Относительный URL нужно сделать абсолютным
            if url.startswith("/"):
                url = f"{self.settings.pypi_mirror_url}{url}"
            elif not url.startswith("http"):
                url = f"{self.settings.pypi_mirror_url}/simple/{normalized_name}/{url}"

            links.append({"url": url, "filename": filename})

        # Фильтрация по версии, если указана
        if version:
            # Имя файла wheel обычно имеет формат: {name}-{version}-{python}-{abi}-{platform}.whl
            version_tag = f"-{version}-"
            links = [link for link in links if version_tag in link["filename"]]

        logger.info("Found package versions", package=package_name, count=len(links))
        return links

    def select_best_wheel(self, urls: List[Dict[str, str]], python_version: str = "cp314") -> Optional[Dict[str, str]]:
        """
        Выбор наиболее подходящего wheel-файла из списка.
        Приоритет: специфичный для платформы/версии Python > универсальный (py3-none-any).
        """
        if not urls:
            return None

        # Фильтруем только .whl файлы
        wheels = [u for u in urls if u["filename"].endswith(".whl")]
        if not wheels:
            return None

        # Ищем сначала специфичный для нужной версии Python (например, cp314)
        for w in wheels:
            if python_version in w["filename"]:
                return w

        # Ищем универсальный (pure python)
        for w in wheels:
            if "py3-none-any" in w["filename"]:
                return w

        # Возвращаем первый попавшийся, если ничего не подошло
        return wheels[0]

            version: str,
            python_version: str = "cp314",

        Скачивание пакета и сохранение в wheel-cache.

        Args:
            package_name: Имя пакета.
            version: Версия.

        Returns:
            Кортеж (путь_к_файлу, sha256_хэш).
        """
        cache_dir = cache_dir or Path(self.settings.wheel_cache_path)
        cache_dir.mkdir(parents=True, exist_ok=True)

        # 1. Проверяем, есть ли уже файл в кэше
        urls = await self.get_package_urls(package_name, version)
        best_match = self.select_best_wheel(urls, python_version)

        if not best_match:
            raise PyPIError(f"No suitable wheel found for {package_name}=={version} (python={python_version})")

        filename = best_match["filename"]
        dest_path = cache_dir / filename

        if dest_path.exists():
            logger.info("Wheel already cached", package=package_name, version=version, path=str(dest_path))
            sha256 = self._calculate_sha256(dest_path)
            return dest_path, sha256

        # 2. Скачиваем файл
        client = self._get_client()
        logger.info("Downloading wheel", package=package_name, version=version, url=best_match["url"])

        try:
            async with client.stream("GET", best_match["url"]) as resp:
                resp.raise_for_status()

                # Скачиваем во временный файл, чтобы не испортить кэш при обрыве
                tmp_path = dest_path.with_suffix(".tmp")
                sha256_hash = hashlib.sha256()

                with open(tmp_path, "wb") as f:
                    async for chunk in resp.aiter_bytes(chunk_size=8192):
                        f.write(chunk)
                        sha256_hash.update(chunk)

                # Переименовываем в целевой файл (атомарно в пределах одной ФС)
                tmp_path.rename(dest_path)

        except httpx.HTTPError as e:
            if tmp_path.exists():
                tmp_path.unlink()
            raise PyPIError(f"Download failed for {package_name}: {e}")

        sha256 = sha256_hash.hexdigest()
        logger.info("Wheel cached successfully", path=str(dest_path), sha256=sha256)

        return dest_path, sha256

    @staticmethod
    def _calculate_sha256(file_path: Path) -> str:
        """Вычисление SHA256 хэша существующего файла."""
        sha256_hash = hashlib.sha256()
        with open(file_path, "rb") as f:
            for chunk in iter(lambda: f.read(8192), b""):
                sha256_hash.update(chunk)
        return sha256_hash.hexdigest()

    @staticmethod
    def _create_tar(files: Dict[str, bytes]) -> bytes:
        """Создает in-memory tar-архив из словаря файлов."""
        buf = io.BytesIO()
        with tarfile.open(fileobj=buf, mode="w") as tar:
            for name, content in files.items():
                info = tarfile.TarInfo(name=name)
                info.size = len(content)
                info.mode = 0o644
                info.uid = 10000
                info.gid = 10000
                tar.addfile(info, io.BytesIO(content))
        return buf.getvalue()

        """Извлекает файлы из tar-архива в словарь."""
        result = {}
        buf = io.BytesIO(tar_bytes)
        with tarfile.open(fileobj=buf, mode="r") as tar:
            for member in tar.getmembers():
                if member.isfile():
                    f = tar.extractfile(member)
                    if f:
                        result[member.name] = f.read()
        return result

    @staticmethod
    def _demux_logs(raw: bytes) -> Tuple[str, str]:
        """
        Разделяет мультиплексированный лог Docker API.
        Формат: [stream_type(1 byte)][0,0,0][size(4 bytes big-endian)][payload]
        stream_type: 1=stdout, 2=stderr
        """
        stdout, stderr = [], []
        idx = 0
        while idx + 8 <= len(raw):
            stream_type = raw[idx]
            size = int.from_bytes(raw[idx + 4: idx + 8], "big")
            idx += 8
            chunk = raw[idx: idx + size].decode("utf-8", errors="replace")
            if stream_type == 1:
                stdout.append(chunk)
            elif stream_type == 2:
                stderr.append(chunk)
            idx += size
        return "".join(stdout), "".join(stderr)
